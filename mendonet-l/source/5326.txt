  set_tris_d(LCD_READ);
  lcd.rw=1; delay_cycles(1);
  lcd.enable=1; delay_cycles(1);
  high=lcd.data;

  lcd.enable=1; delay_us(1);
  low=lcd.data;

  lcd.enable=0;
  set_tris_d(LCD_WRITE);
  return((high<<4)|low);
}
